// --- SOURCE_FILE: ./shared/proto/common.proto ---
syntax = "proto3";
package aegis.common; // 修改包名，更清晰

option go_package = "github.com/Fillped-726/AegisEngine/shared/pb/common"; 

// 基础 3D 向量 (复用性设计)
message PBVector3 {
  float x = 1;
  float y = 2;
  float z = 3;
}

enum PlayerState {
    IDLE = 0;
    MOVE = 1;
    ATTACK = 2;
    DEAD = 3;
}

// 完整的玩家快照信息
message PBPlayerInfo {
  uint64 entity_id = 1;
  PBVector3 pos = 2;
  float direction = 3;
  string name = 4;
  int32 skin_id = 5;
  int32 hp = 6;
  PlayerState state = 7;
  uint32 entity_type = 8;// 0: Player, 1: NPC, 2: Monster, 3: Boss
}


// --- SOURCE_FILE: ./shared/proto/cs_battle.proto ---
syntax = "proto3";
package aegis.cs.battle;

import "common.proto"; // 引用 common

option go_package = "github.com/Fillped-726/AegisEngine/shared/pb/cs_battle";

// [CS] 请求移动
message CSMoveReq {
  aegis.common.PBVector3 target_pos = 1;
  float direction = 2;
  int64 timestamp = 3;
}

// [SC] 移动通知

message SCMoveNtfBatch {
  message MoveInfo {
    uint64 entity_id = 1;
    aegis.common.PBVector3 pos = 2;
    float direction = 3;
  }
  repeated MoveInfo moves = 1; // 一次下发视野内所有移动的玩家
}

// [SC] 进入视野
message SCEnterViewNtf {
  repeated aegis.common.PBPlayerInfo entities = 1;
}

// [SC] 离开视野
message SCLeaveViewNtf {
  repeated uint64 entity_ids = 1;
}

// [CS] 请求更新状态
message CSUpdateStateReq {
  aegis.common.PlayerState state = 1;
}

// [SC] 状态更新
message SCStateUpdateBatch {
  message StateUpdateInfo {
    uint64 entity_id = 1;
    aegis.common.PlayerState state = 2;
  }
  repeated StateUpdateInfo updates = 1;
}

// [CS] 请求释放技能
message CSSkillCastReq {
  uint32 skill_id = 1;
  uint64 target_id = 2; // 如果是无锁定技能（如发波），可传 0
  aegis.common.PBVector3 target_pos = 3; // 施法朝向或目标点
  float direction = 4;
}

// [SC] 伤害事件广播 (用于客户端飘字、受击/死亡表现)
// 注意：伤害是事件流(Event)，不是状态流(State)，必须独立发包保证不丢飘字
message SCDamageNtf {
  uint64 attacker_id = 1;
  uint64 target_id = 2;
  uint32 skill_id = 3;
  int32 damage = 4;      // 造成的伤害数值
  int32 current_hp = 5;  // 扣减后的剩余血量
  bool is_crit = 6;      // 是否暴击 (预留表现位)
}
